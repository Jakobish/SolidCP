**Implement Service:**

```csharp
// File: Data/Services/UserService.cs
using System.Collections.Concurrent;
using SolidCP.WebPortal.Blazor.Data.DTOs;

namespace SolidCP.WebPortal.Blazor.Data.Services
{
    public class UserService : IUserService
    {
        // ... existing implementation ...

        private static readonly ConcurrentDictionary<int, SpaceDetailsDto> _mockSpaces = new()
        {
            [1] = new SpaceDetailsDto
            {
                SpaceId = 1,
                Name = "Default Space",
                Description = "Main organizational space",
                Status = "Active",
                Organization = "SolidCP Organization"
            },
            [2] = new SpaceDetailsDto
            {
                SpaceId = 2,
                Name = "Development Space",
                Description = "Development and testing environment",
                Status = "Active",
                Organization = "SolidCP Development"
            }
        };

        public Task<SpaceDetailsDto?> GetSpaceDetailsAsync(int spaceId)
        {
            return Task.FromResult(_mockSpaces.TryGetValue(spaceId, out var space) ? space : null);
        }

        public Task<bool> UpdateSpaceAsync(SpaceDetailsDto space)
        {
            if (space == null) return Task.FromResult(false);

            space.LastModified = DateTime.UtcNow;
            _mockSpaces[space.SpaceId] = space;
            
            return Task.FromResult(true);
        }

        public Task<List<SpaceSummaryDto>> GetSpaceListAsync()
        {
            var summaries = _mockSpaces.Values.Select(s => new SpaceSummaryDto
            {
                SpaceId = s.SpaceId,
                Name = s.Name,
                Status = s.Status,
                CreatedDate = s.CreatedDate
            }).ToList();

            return Task.FromResult(summaries);
        }
    }

    public class SpaceSummaryDto
    {
        public int SpaceId { get; set; }
        public string Name { get; set; } = string.Empty;
        public string Status { get; set; } = string.Empty;
        public DateTime CreatedDate { get; set; }
    }
}
```

@code {
    [Parameter] public int? SpaceId { get; set; }

    private SpaceDetailsDto? space;
    private bool isLoading = true;
    private bool isSaving = false;

    protected override async Task OnInitializedAsync()
    {
        await LoadSpace();
    }

    protected override async Task OnParametersSetAsync()
    {
        if (SpaceId != null)
        {
            await LoadSpace();
        }
    }

    private async Task LoadSpace()
    {
        isLoading = true;
        try
        {
            if (SpaceId.HasValue)
            {
                space = await UserService.GetSpaceDetailsAsync(SpaceId.Value);
            }
            else
            {
                // Create new space
                space = new SpaceDetailsDto { Status = "Active" };
            }
        }
        catch (Exception ex)
        {
            await NotificationService.ShowError($"Error loading space: {ex.Message}");
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task SaveSpace()
    {
        if (space == null) return;

        isSaving = true;
        try
        {
            bool success = await UserService.UpdateSpaceAsync(space);
            if (success)
            {
                await NotificationService.ShowSuccess("Space updated successfully!");
                await LoadSpace(); // Refresh data
            }
            else
            {
                await NotificationService.ShowError("Failed to update space. Please try again.");
            }
        }
        catch (Exception ex)
        {
            await NotificationService.ShowError($"Error saving space: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private void NavigateBack()
    {
        Navigation.NavigateTo("/spaces");
    }
}

## 🚨 Troubleshooting Guide

### Build Errors

## "Cannot resolve symbol" errors**

- Add `@using SolidCP.WebPortal.Blazor.Data.Services`
- Add `@using SolidCP.WebPortal.Blazor.Data.DTOs`
- Add `@using SolidCP.WebPortal.Blazor.Components`

## "Component not found" errors**

- Verify file exists in `Components/` directory
- Check component name matches exactly
- Ensure component is in correct namespace

## "Service not registered" errors**

- Add service registration to `Program.cs`
- Use exact interface name: `builder.Services.AddScoped<ISpaceService, SpaceService>();`

### Runtime Errors

## Null reference exceptions**

- Always null-check: `if (space != null) { /* use space */ }`
- Initialize collections: `private List<ItemDto> items = new();`

## Navigation issues**

- Use `@inject NavigationManager Navigation`
- Navigate: `Navigation.NavigateTo("/spaces");`
- Include parameters: `Navigation.NavigateTo($"/spaces/details/{spaceId}");`

### UI Issues

## Dark mode not working**

- Always include `dark:` variants: `dark:bg-gray-800`
- Check theme classes: `bg-gray-100 dark:bg-gray-800`

## Responsive layout broken**

- Use Tailwind responsive prefixes: `md:grid-cols-2 lg:grid-cols-3`
- Test by resizing browser window

## Components not styling properly**

- Verify Tailwind CSS is compiled: `npm run build:css`
- Check component CSS classes are Tailwind utilities

## 📝 File Creation Order

1. **DTO** (`Data/DTOs/ClassNameDto.cs`)
2. **Interface** (extend existing interface or create new)
3. **Service Implementation** (extend or modify existing service)
4. **Blazor Page** (`Pages/Folder/ClassName.razor`)
5. **Navigation** (update `NavMenu.razor`)
6. **Program.cs** (register service)
7. **Test** (build and run)
